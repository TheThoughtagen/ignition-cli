Replace the inline-BOLD-per-title implementation with idiomatic Tabs selection:
```rust
let titles: Vec<Line> = Screen::ALL
    .iter()
    .map(|screen| Line::from(screen.title().to_string()))
    .collect();
let active = Screen::ALL
    .iter()
    .position(|screen| *screen == state.screen)
    .unwrap_or(0);
let tabs = Tabs::new(titles)
    .select(active)
    .highlight_style(
        Style::default()
            .fg(state.palette.emphasis)
            .add_modifier(Modifier::BOLD),
    );
frame.render_widget(tabs, area);
```
